# Multi-Stream Recorder — Python Dependencies
# Install with: pip install -r requirements.txt

# ── Required ──────────────────────────────────────
yt-dlp[default]>=2024.01.01   # Stream detection & recording (YouTube, Rumble, TikTok, custom URLs).
                              # [default] pulls in yt-dlp-ejs, needed for YouTube's JS challenge —
                              # without it YouTube recordings drop out every ~15 seconds.
streamlink>=8.0.0             # Kick and Twitch detection & recording. 8.x includes the built-in
                              # Cloudflare JS challenge solver that Kick requires.

# ── Recommended ───────────────────────────────────
psutil>=5.9.0             # Clean process shutdown & orphan cleanup
curl_cffi>=0.10.0         # Browser impersonation for Cloudflare-protected sites (Rumble);
                          # Fishtank.live API (HTTP/3); Kick push channel-ID lookup
websocket-client>=1.6.0   # Kick push notifications — recordings start within seconds of
                          # go-live via Kick's WebSocket event feed (v1.7.0+)

# ── Optional (GUI enhancements) ──────────────────
pystray>=0.19.0           # System tray icon (minimize to tray)
Pillow>=9.0.0             # Required by pystray for tray icon rendering
plyer>=2.1.0              # Desktop notifications (stream started/stopped)

# ── External tools (not pip-installable) ──────────
# ffmpeg — required for remuxing .ts → .mp4
#   Download: https://www.gyan.dev/ffmpeg/builds/ (Windows)
#   Or: sudo apt install ffmpeg (Linux) / brew install ffmpeg (macOS)
#   Must be available in PATH
#
# Deno — recommended for YouTube recording reliability
#   yt-dlp uses Deno to solve YouTube's JS-based n-challenge. Without it,
#   YouTube recordings will drop out roughly every 15 seconds.
#   Download: https://deno.com/
#   Must be available in PATH
